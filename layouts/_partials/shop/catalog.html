{{- /* Liest Shop-Einstellungen und Produkte aus dem Front Matter von content/shop.md, prüft sie und gibt eine einheitliche Struktur zurück (Preise in Cent). Fehler brechen den Hugo-Build mit einer verständlichen Meldung ab. */ -}} {{- $s := .Params.shop | default dict -}} {{- $products := slice -}} {{- $seen := dict -}} {{- range $i, $p := .Params.produkte -}} {{- $id := $p.id | default "" | string -}} {{- if not (findRE `^[a-z0-9][a-z0-9-]*$` $id) -}} {{- errorf "shop.md: Produkt Nr. %d hat keine gültige id (nur a-z, 0-9, -): %q" (add $i 1) $id -}} {{- end -}} {{- if index $seen $id -}} {{- errorf "shop.md: Produkt-id %q ist doppelt" $id -}} {{- end -}} {{- $seen = merge $seen (dict $id true) -}} {{- if not $p.name -}}{{- errorf "shop.md: Produkt %q hat keinen name" $id -}}{{- end -}} {{- /* Fester Preis: preis. Frei wählbar: preis_min + preis_max (preis = Vorschlag, optional) */ -}} {{- $min := float ($p.preis_min | default 0) -}} {{- $max := float ($p.preis_max | default 0) -}} {{- $flexible := or (gt $min 0.0) (gt $max 0.0) -}} {{- $price := float ($p.preis | default $min) -}} {{- if $flexible -}} {{- if or (le $min 0.0) (le $max 0.0) -}}{{- errorf "shop.md: Produkt %q braucht preis_min UND preis_max" $id -}}{{- end -}} {{- if lt $max $min -}}{{- errorf "shop.md: Produkt %q: preis_max ist kleiner als preis_min" $id -}}{{- end -}} {{- if or (lt $price $min) (gt $price $max) -}}{{- errorf "shop.md: Produkt %q: preis (Vorschlag) liegt nicht zwischen preis_min und preis_max" $id -}}{{- end -}} {{- end -}} {{- if le $price 0.0 -}}{{- errorf "shop.md: Produkt %q hat keinen gültigen preis" $id -}}{{- end -}} {{- $step := float ($p.preis_schritt | default 1) -}} {{- if le $step 0.0 -}}{{- errorf "shop.md: Produkt %q: preis_schritt muss größer als 0 sein" $id -}}{{- end -}} {{- $available := true -}} {{- if isset $p "verfuegbar" -}}{{- $available = $p.verfuegbar -}}{{- end -}} {{- $products = $products | append (dict "id" $id "name" ($p.name | string) "price" $price "price_cents" (int (math.Round (mul $price 100))) "flexible" $flexible "min" $min "max" $max "min_cents" (cond $flexible (int (math.Round (mul $min 100))) 0) "max_cents" (cond $flexible (int (math.Round (mul $max 100))) 0) "step_cents" (int (math.Round (mul $step 100))) "available" $available "image" ($p.bild | default "") "photographer" ($p.fotograf | default "") "image_description" ($p.bildbeschreibung | default "") "description" ($p.beschreibung | default "") ) -}} {{- end -}} {{- return dict "currency" ($s.waehrung | default "EUR") "shipping_cents" (int (math.Round (mul (float ($s.versandkosten | default 0)) 100))) "free_shipping_from_cents" (int (math.Round (mul (float ($s.versandkostenfrei_ab | default 0)) 100))) "shipping_countries" ($s.lieferlaender | default (slice "DE")) "delivery" ($s.lieferzeit | default "") "tax_note" ($s.steuerhinweis | default "") "products" $products -}}
